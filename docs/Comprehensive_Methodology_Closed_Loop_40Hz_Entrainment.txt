Phase 3: Phase-Amplitude Coupling (PAC) Label Computation
PAC measures the coupling between the phase of a low-frequency oscillation (theta: 4-8 Hz) and the amplitude of a high-frequency oscillation (gamma: 38-42 Hz). Strong theta-gamma PAC is associated with memory encoding and cognitive function, and is disrupted in Alzheimer's disease.
3.1 PAC Computation Method: Modulation Index (MI)
We use the Modulation Index (MI) method based on Tort et al. (2010), which quantifies PAC using Kullback-Leibler divergence:
Bandpass filter the signal into theta (4-8 Hz) and gamma (38-42 Hz) bands using 4th-order Butterworth filters
Extract instantaneous phase of theta using Hilbert transform
Extract instantaneous amplitude of gamma using Hilbert transform
Bin theta phases into 18 bins of 20 degrees each
Compute mean gamma amplitude in each phase bin
Calculate MI using KL divergence from uniform distribution
3.2 PAC Computation Implementation
from tensorpac import Pac
import numpy as np
from scipy.signal import butter, filtfilt, hilbert

def compute_pac_mi(eeg_window, fs=250):
    """
    Compute Modulation Index PAC for a single EEG window.

    Args:
        eeg_window: (n_channels, n_samples) array
        fs: Sampling frequency (250 Hz)

    Returns:
        pac_value: Mean PAC across channels (scalar)
    """
    # Initialize PAC object with Modulation Index method
    p = Pac(idpac=(2, 0, 0),  # (2, 0, 0) = Modulation Index
            f_pha=(4, 8, 1, 0.5),   # Theta: 4-8 Hz
            f_amp=(38, 42, 1, 0.5))  # Gamma: 38-42 Hz

    # Compute PAC: output shape (n_amp, n_pha, n_epochs, n_channels)
    # We use single epoch, so squeeze appropriately
    xpac = p.filterfit(fs, eeg_window, n_jobs=1)

    # Average across frequency pairs and channels
    pac_value = np.mean(xpac)

    return pac_value

def compute_pac_manual(eeg_window, fs=250, n_bins=18):
    """
    Manual PAC computation using Butterworth filters and Hilbert transform.
    Provides more control over filtering parameters.
    """
    n_channels, n_samples = eeg_window.shape
    pac_values = []

    for ch in range(n_channels):
        signal = eeg_window[ch, :]

        # Bandpass filter for theta phase
        b_theta, a_theta = butter(4, [4, 8], btype='band', fs=fs)
        theta_filt = filtfilt(b_theta, a_theta, signal)
        theta_phase = np.angle(hilbert(theta_filt))

        # Bandpass filter for gamma amplitude
        b_gamma, a_gamma = butter(4, [38, 42], btype='band', fs=fs)
        gamma_filt = filtfilt(b_gamma, a_gamma, signal)
        gamma_amp = np.abs(hilbert(gamma_filt))

        # Compute Modulation Index
        phase_bins = np.linspace(-np.pi, np.pi, n_bins + 1)
        mean_amp = np.zeros(n_bins)

        for i in range(n_bins):
            mask = (theta_phase >= phase_bins[i]) & (theta_phase < phase_bins[i+1])
            if np.sum(mask) > 0:
                mean_amp[i] = np.mean(gamma_amp[mask])

        # Normalize to probability distribution
        mean_amp_norm = mean_amp / (np.sum(mean_amp) + 1e-10)

        # Compute KL divergence from uniform
        uniform = np.ones(n_bins) / n_bins
        kl_div = np.sum(mean_amp_norm * np.log((mean_amp_norm + 1e-10) / uniform))
        mi = kl_div / np.log(n_bins)  # Normalized MI

        pac_values.append(mi)

    return np.mean(pac_values)
3.3 Label Generation Pipeline
def generate_dataset(raw, window_sec=2.0, hop_sec=1.0, fs=250):
    """
    Generate windowed EEG samples with PAC labels.

    Returns:
        X: (n_samples, n_channels, n_timepoints) - EEG windows
        y: (n_samples,) - PAC values (continuous)
    """
    data = raw.get_data()  # (n_channels, n_total_samples)
    window_samples = int(window_sec * fs)  # 500 samples
    hop_samples = int(hop_sec * fs)  # 250 samples

    X, y = [], []

    for start in range(0, data.shape[1] - window_samples, hop_samples):
        window = data[:, start:start + window_samples]
        pac = compute_pac_mi(window, fs)

        X.append(window)
        y.append(pac)

    return np.array(X), np.array(y)

EEGNet is a compact convolutional neural network designed specifically for EEG-based brain-computer interfaces. It uses depthwise and separable convolutions to efficiently learn temporal and spatial features from multi-channel EEG data.

4.2 PyTorch Implementation
import torch
import torch.nn as nn

class EEGNet(nn.Module):
    """
    EEGNet for PAC regression from 7-channel EEG windows.

    Input: (batch, 1, 7, 500) - 7 channels, 500 samples (2 sec @ 250 Hz)
    Output: (batch, 1) - Predicted PAC value
    """
    def __init__(self, n_channels=7, n_samples=500, F1=8, D=2, F2=16, dropout=0.5):
        super(EEGNet, self).__init__()

        # Block 1: Temporal convolution
        self.conv1 = nn.Conv2d(1, F1, (1, 64), padding=(0, 32), bias=False)
        self.bn1 = nn.BatchNorm2d(F1)

        # Block 1: Depthwise spatial convolution
        self.depthwise = nn.Conv2d(F1, F1 * D, (n_channels, 1), groups=F1, bias=False)
        self.bn2 = nn.BatchNorm2d(F1 * D)
        self.elu1 = nn.ELU()
        self.pool1 = nn.AvgPool2d((1, 4))
        self.drop1 = nn.Dropout(dropout)

        # Block 2: Separable convolution
        self.separable1 = nn.Conv2d(F1 * D, F1 * D, (1, 16), padding=(0, 8),
                                     groups=F1 * D, bias=False)
        self.separable2 = nn.Conv2d(F1 * D, F2, (1, 1), bias=False)
        self.bn3 = nn.BatchNorm2d(F2)
        self.elu2 = nn.ELU()
        self.pool2 = nn.AvgPool2d((1, 8))
        self.drop2 = nn.Dropout(dropout)

        # Calculate flattened size
        self.flatten_size = F2 * (n_samples // 32)  # After two pooling layers

        # Regression head
        self.fc = nn.Linear(self.flatten_size, 1)

    def forward(self, x):
        # x: (batch, 1, 7, 500)

        # Block 1
        x = self.conv1(x)
        x = self.bn1(x)
        x = self.depthwise(x)
        x = self.bn2(x)
        x = self.elu1(x)
        x = self.pool1(x)
        x = self.drop1(x)

        # Block 2
        x = self.separable1(x)
        x = self.separable2(x)
        x = self.bn3(x)
        x = self.elu2(x)
        x = self.pool2(x)
        x = self.drop2(x)

        # Output
        x = x.flatten(1)
        x = self.fc(x)

        return x

4.4 Training Loop
def train_eegnet(model, train_loader, val_loader, epochs=100, lr=0.001, patience=15):
    """
    Train EEGNet with early stopping and learning rate scheduling.
    """
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    model = model.to(device)

    criterion = nn.MSELoss()
    optimizer = torch.optim.Adam(model.parameters(), lr=lr, weight_decay=1e-4)
    scheduler = torch.optim.lr_scheduler.ReduceLROnPlateau(
        optimizer, mode='min', factor=0.5, patience=5
    )

    best_val_loss = float('inf')
    patience_counter = 0

    for epoch in range(epochs):
        # Training
        model.train()
        train_loss = 0
        for X_batch, y_batch in train_loader:
            X_batch = X_batch.to(device).float().unsqueeze(1)  # Add channel dim
            y_batch = y_batch.to(device).float().unsqueeze(1)

            optimizer.zero_grad()
            pred = model(X_batch)
            loss = criterion(pred, y_batch)
            loss.backward()
            optimizer.step()
            train_loss += loss.item()

        # Validation
        model.eval()
        val_loss = 0
        with torch.no_grad():
            for X_batch, y_batch in val_loader:
                X_batch = X_batch.to(device).float().unsqueeze(1)
                y_batch = y_batch.to(device).float().unsqueeze(1)
                pred = model(X_batch)
                val_loss += criterion(pred, y_batch).item()

        val_loss /= len(val_loader)
        scheduler.step(val_loss)

        # Early stopping
        if val_loss < best_val_loss:
            best_val_loss = val_loss
            torch.save(model.state_dict(), 'best_eegnet.pth')
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter >= patience:
                print(f"Early stopping at epoch {epoch}")
                break

    model.load_state_dict(torch.load('best_eegnet.pth'))
    return model

Phase 5: Personalization Module
The personalization module adapts the decision thresholds to each individual patient's baseline neural activity, addressing the significant subject-specific variability observed in entrainment responses.
5.1 Rolling Baseline Computation
Maintain a 30-second rolling window of PAC predictions to compute personalized statistics:
Baseline window: 30 seconds = 30 PAC predictions (at 1-second intervals)
Compute running mean (mu) and standard deviation (sigma) of PAC values
Update baseline continuously using circular buffer
5.2 Z-Score Normalization
Convert predicted PAC values to z-scores for threshold comparison:
z = (PAC_predicted - mu_baseline) / sigma_baseline
5.3 Implementation
import numpy as np
from collections import deque

class PersonalizationModule:
    """
    Maintains patient-specific baseline and computes z-scores.
    """
    def __init__(self, window_size=30, min_samples=10):
        """
        Args:
            window_size: Number of seconds for rolling baseline (30s)
            min_samples: Minimum samples before z-score is computed
        """
        self.window_size = window_size
        self.min_samples = min_samples
        self.pac_buffer = deque(maxlen=window_size)

    def update(self, pac_value):
        """Add new PAC value to the rolling buffer."""
        self.pac_buffer.append(pac_value)

    def get_baseline_stats(self):
        """Return mean and std of baseline PAC values."""
        if len(self.pac_buffer) < self.min_samples:
            return None, None

        values = np.array(self.pac_buffer)
        return np.mean(values), np.std(values) + 1e-8  # Avoid division by zero

    def compute_zscore(self, pac_value):
        """
        Compute z-score of current PAC relative to baseline.
        Returns None if insufficient baseline data.
        """
        mu, sigma = self.get_baseline_stats()
        if mu is None:
            return None

        z_score = (pac_value - mu) / sigma
        return z_score

    def reset(self):
        """Clear baseline buffer (e.g., for new patient)."""
        self.pac_buffer.clear()

6.2 State Persistence (Hysteresis)
To prevent rapid oscillation between states, enforce a minimum hold time of 5 seconds before any state change:
Track time since last state change
Only allow state transition if elapsed time >= 5.0 seconds
This provides stability and mimics realistic therapy delivery
6.3 Complete Controller Implementation
from enum import Enum
import time

class StimState(Enum):
    STIMULATE = 1
    REST = 0

class ClosedLoopController:
    """
    Closed-loop controller for 40 Hz auditory entrainment.

    Integrates:
    - EEGNet for PAC prediction
    - Personalization module for baseline adaptation
    - Decision engine with hysteresis
    """
    def __init__(self, model, device='cpu', z_low=-0.5, z_high=0.5,
                 hold_time=5.0, baseline_window=30):
        """
        Args:
            model: Trained EEGNet model
            device: 'cuda' or 'cpu'
            z_low: Z-score threshold for weak coupling
            z_high: Z-score threshold for strong coupling
            hold_time: Minimum seconds before state change
            baseline_window: Seconds for rolling baseline
        """
        self.model = model.to(device)
        self.model.eval()
        self.device = device

        self.z_low = z_low
        self.z_high = z_high
        self.hold_time = hold_time

        self.personalization = PersonalizationModule(window_size=baseline_window)

        self.current_state = StimState.STIMULATE  # Start with stimulation
        self.last_change_time = 0
        self.time_counter = 0

    def predict_pac(self, eeg_window):
        """
        Predict PAC from 2-second EEG window.

        Args:
            eeg_window: (7, 500) numpy array
        Returns:
            pac_pred: Predicted PAC value
        """
        with torch.no_grad():
            x = torch.tensor(eeg_window).float().unsqueeze(0).unsqueeze(0)
            x = x.to(self.device)  # (1, 1, 7, 500)
            pac_pred = self.model(x).item()
        return pac_pred

    def step(self, eeg_window):
        """
        Execute one control step.

        Args:
            eeg_window: (7, 500) numpy array - 2 seconds of 7-channel EEG

        Returns:
            action: StimState (STIMULATE or REST)
            pac_pred: Predicted PAC value
            z_score: Z-score relative to baseline (or None if insufficient data)
        """
        # Predict PAC
        pac_pred = self.predict_pac(eeg_window)

        # Update personalization baseline
        self.personalization.update(pac_pred)

        # Compute z-score
        z_score = self.personalization.compute_zscore(pac_pred)

        # Increment time counter
        self.time_counter += 1  # Each step = 1 second

        # Decision logic with hysteresis
        if z_score is not None:
            time_since_change = self.time_counter - self.last_change_time

            if time_since_change >= self.hold_time:
                new_state = None

                if z_score < self.z_low:
                    new_state = StimState.STIMULATE
                elif z_score > self.z_high:
                    new_state = StimState.REST
                # else: maintain current state

                if new_state is not None and new_state != self.current_state:
                    self.current_state = new_state
                    self.last_change_time = self.time_counter

        return self.current_state, pac_pred, z_score

    def reset(self):
        """Reset controller state for new session."""
        self.personalization.reset()
        self.current_state = StimState.STIMULATE
        self.last_change_time = 0
        self.time_counter = 0

Phase 7: Simulation-Based Validation
Since we cannot conduct live experiments, we validate the closed-loop controller through simulation using the recorded EEG data, modeling the brain's response dynamics during stimulation and rest periods.
7.1 Response Dynamics Modeling
Extract characteristic PAC response patterns from the dataset:
Segment data by condition (40s stimulus periods vs 20s rest periods)
Compute PAC time series for each condition across all subjects
Model PAC rise rate during stimulation (tau_rise)
Model PAC decay rate during rest (tau_decay)
Apply exponential dynamics: PAC(t+1) = PAC(t) + alpha * (target - PAC(t))
7.2 Simulation Framework
class EntrainmentSimulator:
    """
    Simulates brain response to closed-loop stimulation decisions.
    """
    def __init__(self, tau_rise=0.15, tau_decay=0.10,
                 pac_max=0.3, pac_min=0.05, noise_std=0.02):
        """
        Args:
            tau_rise: Rate of PAC increase during stimulation
            tau_decay: Rate of PAC decrease during rest
            pac_max: Maximum achievable PAC
            pac_min: Minimum PAC during rest
            noise_std: Gaussian noise standard deviation
        """
        self.tau_rise = tau_rise
        self.tau_decay = tau_decay
        self.pac_max = pac_max
        self.pac_min = pac_min
        self.noise_std = noise_std
        self.current_pac = pac_min

    def step(self, action):
        """
        Simulate one timestep of brain response.

        Args:
            action: StimState.STIMULATE or StimState.REST

        Returns:
            pac: Simulated PAC value after this timestep
        """
        if action == StimState.STIMULATE:
            target = self.pac_max
            tau = self.tau_rise
        else:
            target = self.pac_min
            tau = self.tau_decay

        # Exponential approach to target
        self.current_pac += tau * (target - self.current_pac)

        # Add noise
        self.current_pac += np.random.normal(0, self.noise_std)
        self.current_pac = np.clip(self.current_pac, 0, 1)

        return self.current_pac

    def reset(self, initial_pac=None):
        """Reset simulator state."""
        self.current_pac = initial_pac if initial_pac else self.pac_min

7.6 Complete Validation Code
def run_validation(controller, simulator, test_data, duration_sec=360):
    """
    Run closed-loop simulation and compare to fixed schedule.

    Args:
        controller: ClosedLoopController instance
        simulator: EntrainmentSimulator instance
        test_data: Test subject EEG windows
        duration_sec: Total simulation duration

    Returns:
        results: Dict with metrics
    """
    controller.reset()
    simulator.reset()

    closed_loop_pacs = []
    decisions = []

    # Run closed-loop simulation
    for t in range(duration_sec):
        # Get EEG window (use test data cyclically)
        eeg_idx = t % len(test_data)
        eeg_window = test_data[eeg_idx]

        # Controller decision
        action, pac_pred, z_score = controller.step(eeg_window)

        # Simulate brain response
        actual_pac = simulator.step(action)

        closed_loop_pacs.append(actual_pac)
        decisions.append(action.value)

    # Run fixed-schedule simulation
    simulator.reset()
    fixed_pacs = []

    for t in range(duration_sec):
        cycle_pos = t % 60  # 40s stim + 20s rest
        if cycle_pos < 40:
            action = StimState.STIMULATE
        else:
            action = StimState.REST

        actual_pac = simulator.step(action)
        fixed_pacs.append(actual_pac)

    # Compute metrics
    closed_mean = np.mean(closed_loop_pacs)
    fixed_mean = np.mean(fixed_pacs)
    improvement = (closed_mean - fixed_mean) / fixed_mean * 100

    stim_time_closed = np.mean(decisions) * 100
    stim_time_fixed = 40/60 * 100

    efficiency_closed = closed_mean / (stim_time_closed / 100)
    efficiency_fixed = fixed_mean / (stim_time_fixed / 100)

    return {
        'closed_loop_mean_pac': closed_mean,
        'fixed_schedule_mean_pac': fixed_mean,
        'pac_improvement_percent': improvement,
        'closed_loop_stim_time_percent': stim_time_closed,
        'efficiency_ratio': efficiency_closed / efficiency_fixed
    }
